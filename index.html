💻 Developer Quickstart

Integrate sub-millisecond video routing into your data ingestion pipeline.

Python 3.10+ • PyTorch pip install visionchunk 
import torch
from visionchunk import VideoRouter

# Initialize lightweight router (loads 351k weights directly to CPU)
router = VideoRouter.from_pretrained("harsh2b/visionchunk", device="cpu")

# Classify video stream and get optimal chunking policy in <1 ms
result = router.dispatch("input_stream.mp4")

print(f"Domain: {result.category} ({result.confidence:.1%})")
print(f"Policy: {result.policy.name} | Window: {result.policy.window_seconds}s")
print(f"Latency: {result.latency_ms:.2f} ms on CPU")
